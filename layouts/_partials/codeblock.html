{{- /* Shared validated Chroma renderer for shortcodes, concepts, and fences. */ -}} {{- $code := .code | strings.TrimSpace -}} {{- $lang := .lang | default "" | strings.TrimSpace | strings.ToLower -}} {{- $position := .position | default (.page.Path | default "unknown page") -}} {{- if eq $lang "" }}{{ errorf "%s: code block has no language" $position }}{{ end -}} {{- if not (transform.CanHighlight $lang) }}{{ errorf "%s: Chroma does not support code language %q" $position $lang }}{{ end -}} {{- $class := "code" -}} {{- with .classes }}{{ $class = printf "%s %s" $class . }}{{ end -}} {{- $highlighted := "" -}} {{- if .context -}} {{- $options := merge (dict "noClasses" false "wrapperClass" $class) (.options | default dict) -}} {{- $result := transform.HighlightCodeBlock .context $options -}} {{- $highlighted = $result.Wrapped -}} {{- else -}} {{- if strings.Contains $code "WOMBATCODEHIGHLIGHT" }}{{ errorf "%s: code block contains the renderer's reserved placeholder prefix" $position }}{{ end -}} {{- $restores := slice -}} {{- if .emphasis -}} {{- range $index, $match := findRESubmatch `@@([^\n]+?)@@` $code -}} {{- $placeholder := printf "WOMBATCODEHIGHLIGHT%04dTOKEN" $index -}} {{- $code = replace $code (index $match 0) $placeholder 1 -}} {{- $restores = $restores | append (dict "placeholder" $placeholder "text" (index $match 1)) -}} {{- end -}} {{- if strings.Contains $code "@@" }}{{ errorf "%s: code block contains an unmatched @@ emphasis marker" $position }}{{ end -}} {{- end -}} {{- $highlighted = transform.Highlight $code $lang (dict "noClasses" false "wrapperClass" $class) -}} {{- range $restores -}} {{- $replacement := printf `%s` (.text | htmlEscape) -}} {{- $highlighted = replace $highlighted .placeholder $replacement -}} {{- end -}} {{- end -}} {{- with .filename -}} {{- $highlighted = printf `
%s
%s` (. | htmlEscape) $highlighted -}} {{- end -}} {{- return (safeHTML $highlighted) -}}
