{{- /* Renders a source listing: escape it, then pick out the comments so authors write plain code and never hand-wrap a . Called with: (dict "code" "…" "marker" "//" "size" "small") */ -}} {{- $marker := .marker | default "//" -}} {{- $size := .size | default "" -}} {{- $escaped := .code | strings.TrimSpace | htmlEscape -}} {{- $pattern := printf `(?m)(%s.*)$` (strings.TrimSpace $marker) -}} {{- $coloured := replaceRE $pattern `$1` $escaped -}} {{- $class := cond (ne $size "") (printf "code %s" $size) "code" -}} {{- return printf `
%s
` $class $coloured -}}
