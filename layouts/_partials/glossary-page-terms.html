{{- /* Set of base glossary keys referenced by a content page. */ -}} {{- $source := printf "%s\n%s" .RawContent (.Params | jsonify) -}} {{- $terms := dict -}} {{- range findRE `\[\[[^][]+\]\]` $source -}} {{- $reference := partial "parse-reference.html" (dict "raw" . "position" $.File.Path) -}} {{- if eq $reference.kind "term" }}{{ $terms = merge $terms (dict $reference.base true) }}{{ end -}} {{- end -}} {{- return $terms -}}
