$introductory.Params.project_order -}} {{- if or (not $order) (not (reflect.IsSlice $order)) -}} {{- errorf "%s: project_order must be a list" $introductory.File.Path -}} {{- end -}} {{- $codingTracks := slice (dict "platform" "wombat" "label" "Wombat Coding" "pattern" "/introductory/coding/project-%02d") (dict "platform" "ev3" "label" "EV3 Coding" "pattern" "/introductory/ev3/project-%02d") (dict $position -}} {{- end -}} {{- $systemsReference := printf "/introductory/systems/project-%02d" $systemsNumber -}} {{- $systemsPage := site.GetPage $systemsReference -}} {{- if and (not $allowIncompleteProjectOrder) (not $systemsPage) -}} {{- errorf "%s does not resolve: %q" $position $systemsReference -}} {{- end -}} {{- $codingPages := slice -}} {{- range $codingTracks -}} {{- $codingReference := printf .pattern $codingNumber -}} {{- $codingPage := site.GetPage $codingReference -}} {{- if and (not $allowIncompleteProjectOrder) (not $codingPage) -}} {{- errorf "%s does not resolve: %q" $position $codingReference -}} {{- end -}} {{- $codingPages = $codingPages | append (dict "platform" .platform "label" .label "page" $codingPage "number" $codingNumber ) -}} {{- end -}} {{- $relations = $relations | append (dict "systems_number" $systemsNumber "systems_page" $systemsPage "coding_number" $codingNumber "coding_pages" $codingPages "through_coding_number" $throughNumber "kind" $entry.kind ) -}} {{- end -}} {{- return $relations -}}
