{{- /* One cell renderer for gridtable and repeattable. */ -}} {{- $cell := .cell -}} {{- $column := .column | default dict -}} {{- if eq (index $cell "kind" | default "") "number" -}} {{ index $cell "text" }} {{- else if isset $cell "text" -}} {{ .page.RenderString (printf "%v" (index $cell "text")) }} {{- else -}} {{- $sourceKey := index $cell "key" | default (index $column "key") -}} {{- if not $sourceKey }}{{ errorf "%s: table input cell needs a key" .position }}{{ end -}} {{- $key := $sourceKey -}} {{- if .repeat }}{{ $key = printf "%s%d_%s" .prefix .rowNumber $sourceKey }}{{ end -}} {{- $aria := index $cell "aria" | default (index $column "aria") -}} {{- if not $aria }}{{ $aria = index $column "head" }}{{ end -}} {{- if not $aria }}{{ $aria = .rowLabel }}{{ end -}} {{- if not $aria }}{{ $aria = humanize $sourceKey }}{{ end -}} {{- if .repeat }}{{ $aria = printf "%s %d" $aria .rowNumber }}{{ end -}} {{- end -}}
