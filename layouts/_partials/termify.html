{{- /* Rewrite validated glossary references and mission links. */ -}} {{- $page := .page -}} {{- $content := .content -}} {{- $usedTerms := $page.Store.Get "usedTerms" | default slice -}} {{- range findRE `\[\[[^][]+\]\]` $content -}} {{- $token := . -}} {{- $reference := partial "parse-reference.html" (dict "raw" $token "position" $page.File.Path) -}} {{- $resolved := partial "resolve-reference.html" (dict "reference" $reference "track" ($page.Params.track | default "c") "position" $page.File.Path) -}} {{- if eq $reference.kind "mission" -}} {{- $text := $reference.label | default $resolved.title -}} {{- $href := printf `%s#%s` $resolved.page.RelPermalink $resolved.tier -}} {{- $html := printf `%s` $href ($text | htmlEscape) -}} {{- $content = replace $content $token $html -}} {{- else -}} {{- $text := $reference.label | default $resolved.base -}} {{- $html := printf `` $resolved.lookup $text -}} {{- $content = replace $content $token $html -}} {{- $usedTerms = $usedTerms | append $resolved.lookup -}} {{- end -}} {{- end -}} {{- $page.Store.Set "usedTerms" (uniq $usedTerms) -}} {{- return $content -}}
