{{- /* Rewrite validated glossary and mission references and record page usage. */ -}} {{- $page := .page -}} {{- $content := .content -}} {{- $usedTerms := $page.Store.Get "usedTerms" | default slice -}} {{- $usedMissions := $page.Store.Get "usedMissions" | default slice -}} {{- range findRE `\[\[[^][]+\]\]` $content -}} {{- $token := . -}} {{- $reference := partial "parse-reference.html" (dict "raw" $token "position" $page.File.Path) -}} {{- $resolved := partial "resolve-reference.html" (dict "reference" $reference "track" ($page.Params.track | default "c") "position" $page.File.Path) -}} {{- if eq $reference.kind "mission" -}} {{- $text := $reference.label | default $resolved.title -}} {{- $html := printf `%s` $resolved.number $resolved.tier $text -}} {{- $content = replace $content $token $html -}} {{- $usedMissions = $usedMissions | append $resolved.number -}} {{- else -}} {{- $text := $reference.label | default $resolved.base -}} {{- $html := printf `%s` $resolved.lookup $text -}} {{- $content = replace $content $token $html -}} {{- $usedTerms = $usedTerms | append $resolved.lookup -}} {{- end -}} {{- end -}} {{- $page.Store.Set "usedTerms" (uniq $usedTerms) -}} {{- $page.Store.Set "usedMissions" (uniq $usedMissions) -}} {{- return $content -}}
