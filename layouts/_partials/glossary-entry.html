{{- /* Validate and resolve a parsed term, including sense and track wording. */ -}} {{- $ref := .reference -}} {{- $base := $ref.base -}} {{- $sense := $ref.qualifier -}} {{- $entry := index site.Data.glossary $base -}} {{- if not $entry -}} {{- errorf "%s: [[%s]] is not a term in data/glossary.yaml" .position $ref.target -}} {{- else if and (ne $sense "") (not (isset $entry "senses")) -}} {{- errorf "%s: [[%s]] asks for sense %q but %s has no senses in data/glossary.yaml" .position $ref.target $sense $base -}} {{- else if and (ne $sense "") (not (isset $entry.senses $sense)) -}} {{- errorf "%s: [[%s]] asks for sense %q, which is not defined under %s.senses" .position $ref.target $sense $base -}} {{- end -}} {{- $title := $entry.title -}} {{- $body := cond (and (eq (.track | default "c") "python") (isset $entry "python")) (index $entry "python") $entry.body -}} {{- if ne $sense "" -}} {{- $alternate := index $entry.senses $sense -}} {{- $title = $alternate.title | default $title -}} {{- $body = $alternate.body -}} {{- if and (eq (.track | default "c") "python") (isset $alternate "python") }}{{ $body = index $alternate "python" }}{{ end -}} {{- end -}} {{- $lookup := cond (ne $sense "") (printf "%s:%s" $base $sense) $base -}} {{- return (dict "base" $base "sense" $sense "lookup" $lookup "title" $title "body" $body "entry" $entry) -}}
